5. CPU & MEMORY MANAGEMENT

[root@prod-hypervisor ~]# virsh setvcpus prod-web-01 6 --live
[root@prod-hypervisor ~]# virsh setvcpus prod-web-01 6 --config

# Set max vCPUs (persistent)
[root@prod-hypervisor ~]# virsh setvcpus prod-web-01 8 --maximum --config

# Set memory (live)
[root@prod-hypervisor ~]# virsh setmem prod-web-01 10485760 --live
[root@prod-hypervisor ~]# virsh setmem prod-web-01 10485760 --config

# Set max memory (requires VM restart)
